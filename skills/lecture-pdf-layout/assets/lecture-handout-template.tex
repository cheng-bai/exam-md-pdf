\noindent\textbf{例 1}\quad \difficulty{3}\quad 在这里放入题目文本。
\[
  a^2+b^2\geq 2ab.
\]

\begin{answerbox}
\noindent\textbf{解}\quad 在这里放入解答过程。
\[
  a^2+b^2-2ab=(a-b)^2\geq 0.
\]
因此
\[
  \color{markred}{a^2+b^2\geq 2ab}.
\]
\end{answerbox}
